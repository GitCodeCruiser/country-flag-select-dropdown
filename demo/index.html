Code Examples

// npm install country-flag-select
import { CountryFlagSelect } from 'country-flag-select';

const select = new CountryFlagSelect('#my-select', {
  placeholder: 'Select a country...',
  valueType: 'short',   // 'short' = 'US' | 'long' = 'United States'
  flagType: 'emoji',    // 'emoji' | 'image' | 'none'
  multi: false,
  searchable: true,
  clearable: true,
  theme: 'auto',         // 'auto' | 'light' | 'dark'
  onChange: (value) => {
    console.log('Selected:', value); // 'US'
  },
});

// Programmatic API
select.getValue();         // => 'US'
select.setValue('GB');    // set programmatically
select.reset();           // clear selection
select.destroy();         // clean up DOM
import { CountryFlagSelect } from 'country-flag-select/react';

  return (

      onChange={setCountry}
      placeholder="Select a country..."
      valueType="short"
      multi={false}
      searchable
      clearable
      theme="auto"
    />
  );
}

valueType'short'|'long''short'Return ISO code ('US') or full name ('United States')
flagType'emoji'|'image'|'none''emoji'How to render country flags

placeholderstring'Select a country...'Placeholder text
searchablebooleantrueShow search input in dropdown
clearablebooleantrueShow a clear (×) button

includestring[][]Whitelist specific country codes
excludestring[][]Blacklist specific country codes
